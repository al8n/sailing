# Two groups on one host set elect independently: each campaign + stabilize moves ONLY its own
# group's traffic (the g-scoped stabilize leaves the sibling's bus untouched), different nodes
# end up leading the two groups, and between the rounds the sibling is provably pristine.
create-group g=100 voters=(1,2,3)
----
n1: created g100 voters={1,2,3}
n2: created g100 voters={1,2,3}
n3: created g100 voters={1,2,3}

create-group g=200 voters=(1,2,3)
----
n1: created g200 voters={1,2,3}
n2: created g200 voters={1,2,3}
n3: created g200 voters={1,2,3}

campaign g=100 1
----
g100 n1 candidate term=1
g100 RequestVote term=1 last=0/0 prevote=false
g100 RequestVote term=1 last=0/0 prevote=false

stabilize g=100
----
> n2 recv g100 RequestVote term=1 last=0/0 prevote=false
> n3 recv g100 RequestVote term=1 last=0/0 prevote=false
> n2 ready
  g100 VoteResponse term=1 prevote=false reject=false
> n3 ready
  g100 VoteResponse term=1 prevote=false reject=false
> n1 recv g100 VoteResponse term=1 prevote=false reject=false
> n1 recv g100 VoteResponse term=1 prevote=false reject=false
> n1 ready
  g100 Heartbeat term=1 commit=0
  g100 Heartbeat term=1 commit=0
  g100 AppendEntries term=1 prev=0/0 commit=0 entries=[1/1 empty]
  g100 AppendEntries term=1 prev=0/0 commit=0 entries=[1/1 empty]
  g100 n1 leader-changed term=1 lead=1
> n2 recv g100 Heartbeat term=1 commit=0
> n3 recv g100 Heartbeat term=1 commit=0
> n2 recv g100 AppendEntries term=1 prev=0/0 commit=0 entries=[1/1 empty]
> n3 recv g100 AppendEntries term=1 prev=0/0 commit=0 entries=[1/1 empty]
> n2 ready
  g100 HeartbeatResponse term=1
  g100 AppendResponse term=1 match=1
  g100 n2 leader-changed term=1 lead=1
> n3 ready
  g100 HeartbeatResponse term=1
  g100 AppendResponse term=1 match=1
  g100 n3 leader-changed term=1 lead=1
> n1 recv g100 HeartbeatResponse term=1
> n1 recv g100 AppendResponse term=1 match=1
> n1 recv g100 HeartbeatResponse term=1
> n1 recv g100 AppendResponse term=1 match=1
> n1 ready
  g100 AppendEntries term=1 prev=0/0 commit=0 entries=[1/1 empty]
  g100 AppendEntries term=1 prev=1/1 commit=1 entries=[]
  g100 AppendEntries term=1 prev=0/0 commit=1 entries=[1/1 empty]
  g100 AppendEntries term=1 prev=1/1 commit=1 entries=[]
> n2 recv g100 AppendEntries term=1 prev=0/0 commit=0 entries=[1/1 empty]
> n2 recv g100 AppendEntries term=1 prev=1/1 commit=1 entries=[]
> n3 recv g100 AppendEntries term=1 prev=0/0 commit=1 entries=[1/1 empty]
> n3 recv g100 AppendEntries term=1 prev=1/1 commit=1 entries=[]
> n2 ready
  g100 AppendResponse term=1 match=1
  g100 AppendResponse term=1 match=1
> n3 ready
  g100 AppendResponse term=1 match=1
  g100 AppendResponse term=1 match=1
> n1 recv g100 AppendResponse term=1 match=1
> n1 recv g100 AppendResponse term=1 match=1
> n1 recv g100 AppendResponse term=1 match=1
> n1 recv g100 AppendResponse term=1 match=1
